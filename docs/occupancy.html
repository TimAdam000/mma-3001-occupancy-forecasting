def aggregate_chunk(chunk, bin_freq='15min'): 
40def aggregate_chunk(chunk, bin_freq="15min"):
41    """Aggregate one chunk of raw events to one row per (space, time bin).
42
43    The status kept for a bin is the *last* event in it; headcount is the maximum.
44
45    Parameters
46    ----------
47    chunk : pandas.DataFrame
48        Columns ``floorspaceid``, ``occupancystatus``, ``headcount``, ``collecteddate``.
49    bin_freq : str
50        Pandas frequency string for the bin width.
51
52    Returns
53    -------
54    pandas.DataFrame
55        Columns ``floorspaceid, bin, status, headcount, n_events, t_last``.
56        Rows with an unparseable time or an unknown status are dropped.
57    """
58    t = pd.to_datetime(chunk["collecteddate"], utc=True, errors="coerce")
59    ch = chunk.assign(t=t).dropna(subset=["t"])
60    ch["bin"] = ch["t"].dt.floor(bin_freq)
61    ch["status"] = ch["occupancystatus"].map(STATUS_CODE)
62    ch = ch.dropna(subset=["status"]).sort_values("t")
63    return ch.groupby(["floorspaceid", "bin"]).agg(
64        status=("status", "last"), headcount=("headcount", "max"),
65        n_events=("status", "size"), t_last=("t", "max")).reset_index()

def fill_time_grid(occ, bin_freq='15min', max_ffill_bins=8): 
 87def fill_time_grid(occ, bin_freq="15min", max_ffill_bins=8):
 88    """Put each space on a regular time grid, carrying the last status over short gaps.
 89
 90    A sensor only reports when something changes, so empty bins are normal. The last known
 91    status and headcount are carried forward for at most ``max_ffill_bins`` bins (8 x 15 min
 92    = 2 h). Longer gaps stay empty and are dropped, rather than guessed.
 93
 94    Parameters
 95    ----------
 96    occ : pandas.DataFrame
 97        Output of :func:`combine_chunks`.
 98    bin_freq : str
 99        Bin width.
100    max_ffill_bins : int
101        Maximum number of consecutive empty bins to fill.
102
103    Returns
104    -------
105    pandas.DataFrame
106        Columns ``timestamp, status, headcount, n_events, floorspaceid``.
107    """
108    grids = []
109    for sid, g in occ.groupby("floorspaceid"):
110        full = pd.date_range(g["bin"].min(), g["bin"].max(), freq=bin_freq)
111        g = g.set_index("bin").reindex(full)
112        g.index.name = "timestamp"
113        g["n_events"] = g["n_events"].fillna(0)
114        g[["status", "headcount"]] = g[["status", "headcount"]].ffill(limit=max_ffill_bins)
115        g["floorspaceid"] = sid
116        grids.append(g.reset_index())
117    return pd.concat(grids, ignore_index=True)

def load_occupancy( occ_path, space_to_room, bin_freq='15min', tz='Australia/Melbourne', occupied_states=('CurrentlyOccupied',), max_ffill_bins=8, chunksize=1000000): 
120def load_occupancy(occ_path, space_to_room, bin_freq="15min", tz="Australia/Melbourne",
121                   occupied_states=("CurrentlyOccupied",), max_ffill_bins=8,
122                   chunksize=1_000_000):
123    """Full pipeline: raw event log -> regular 15-minute occupancy table per room.
124
125    Parameters
126    ----------
127    occ_path : str
128        Path to the occupancy CSV.
129    space_to_room : dict
130        ``{SpaceId: room}`` from :func:`load_space_lookup`. Unknown spaces are named
131        ``unmapped_<first 8 characters of the id>``.
132    bin_freq, tz, max_ffill_bins, chunksize
133        Bin width, local time zone, maximum gap to fill, rows read per chunk.
134    occupied_states : iterable of str
135        Statuses that count as occupied (default only ``CurrentlyOccupied``).
136
137    Returns
138    -------
139    pandas.DataFrame
140        Columns include ``timestamp`` (UTC), ``local`` (tz-aware local time), ``room``,
141        ``status`` (0/1/2), ``headcount`` and ``occupied`` (0/1).
142    """
143    reader = pd.read_csv(occ_path,
144                         usecols=["floorspaceid", "occupancystatus", "headcount", "collecteddate"],
145                         chunksize=chunksize)
146    occ = combine_chunks([aggregate_chunk(ch, bin_freq) for ch in reader])
147    occ = fill_time_grid(occ, bin_freq, max_ffill_bins)
148    occ["room"] = occ["floorspaceid"].map(space_to_room).fillna(
149        "unmapped_" + occ["floorspaceid"].str[:8])
150    occ = occ.dropna(subset=["status"])
151    code_to_name = {v: k for k, v in STATUS_CODE.items()}
152    occ["occupied"] = occ["status"].map(code_to_name).isin(set(occupied_states)).astype(int)
153    occ["local"] = occ["timestamp"].dt.tz_convert(tz)
154    return occ

def chronological_split(d, fractions=(0.6, 0.8)): 
63def chronological_split(d, fractions=(0.6, 0.8)):
64    """Split by time, never at random: the model is tested on a later period than it trained on.
65
66    Parameters
67    ----------
68    d : pandas.DataFrame
69        Needs a ``timestamp`` column.
70    fractions : tuple of float

72
73    Returns
74    -------
75    train, val, test : pandas.DataFrame
76    """
77    t1, t2 = d["timestamp"].quantile(list(fractions))

79            d[(d.timestamp > t1) & (d.timestamp <= t2)],

def make_features(occ, horizon_bins=4, bin_freq='15min'): 
11def make_features(occ, horizon_bins=4, bin_freq="15min"):
12    """Add the target and features to a regular occupancy table.
13
14    The target is the occupancy ``horizon_bins`` bins ahead. It is matched on real
15    timestamps, so a gap in the data gives a missing target and the row is dropped, rather
16    than silently pairing the wrong row. Features use the present and past only (no leakage):
17    current state, rolling means over the last 1 h and 4 h, the same time one week earlier
18    (``-1`` if unknown), and sine/cosine of local time of day, day of week and room.
19
20    Parameters
21    ----------
22    occ : pandas.DataFrame
23        Needs columns ``room, timestamp, local, occupied, headcount``.
24    horizon_bins : int
25        Forecast horizon in bins (4 x 15 min = 1 hour).
26    bin_freq : str
27        Bin width, must match the table.
28
29    Returns
30    -------
31    pandas.DataFrame
32        Rows with a known target, with ``target`` and every column in :data:`FEATURES`.
33    """
34    step = pd.Timedelta(bin_freq)
35    d = (occ[["room", "timestamp", "local", "occupied", "headcount"]]
36         .sort_values(["room", "timestamp"]).reset_index(drop=True))
37
38    tgt = d[["room", "timestamp", "occupied"]].copy()
39    tgt["timestamp"] -= horizon_bins * step
40    d = d.merge(tgt.rename(columns={"occupied": "target"}), on=["room", "timestamp"], how="left")
41
42    wk = d[["room", "timestamp", "occupied"]].copy()
43    wk["timestamp"] += pd.Timedelta(days=7)
44    d = d.merge(wk.rename(columns={"occupied": "occ_last_week"}), on=["room", "timestamp"], how="left")
45
46    d["occ_mean_1h"] = d.groupby("room")["occupied"].transform(lambda s: s.rolling(4, min_periods=1).mean())
47    d["occ_mean_4h"] = d.groupby("room")["occupied"].transform(lambda s: s.rolling(16, min_periods=1).mean())
48
49    hr = d["local"].dt.hour + d["local"].dt.minute / 60
50    d["hour_sin"] = np.sin(2 * np.pi * hr / 24)
51    d["hour_cos"] = np.cos(2 * np.pi * hr / 24)
52    d["hour"] = d["local"].dt.hour
53    d["dayofweek"] = d["local"].dt.dayofweek
54    d["is_weekend"] = (d["dayofweek"] >= 5).astype(int)
55    d["room_code"] = d["room"].astype("category").cat.codes
56
57    d = d.dropna(subset=["target"] + [f for f in FEATURES if f != "occ_last_week"]).reset_index(drop=True)
58    d["target"] = d["target"].astype(int)
59    d["occ_last_week"] = d["occ_last_week"].fillna(-1)
60    return d

The target is the occupancy horizon_bins bins ahead. It is matched on real timestamps, so a gap in the data gives a missing target and the row is dropped, rather than silently pairing the wrong row. def feature_group_sensitivity(train, val, features, groups, params, random_state=42): 
 89def feature_group_sensitivity(train, val, features, groups, params, random_state=42):
 90    """Retrain with one group of features removed at a time, scored on validation.
 91
 92    Parameters
 93    ----------
 94    groups : dict
 95        ``{group name: [columns]}``.
 96    params : dict
 97        Random Forest settings (``n_estimators, max_depth, min_samples_leaf``).
 98
 99    Returns
100    -------
101    pandas.DataFrame
102        ``removed, val_F1, change`` (vs all features) and ``fit_s``.
103    """
104    def fit_eval(feats):
105        t0 = time.time()
106        m = RandomForestClassifier(**params, n_jobs=-1, random_state=random_state).fit(train[feats], train["target"])
107        return f1_score(val["target"], m.predict(val[feats])), time.time() - t0
108
109    full_f1, full_t = fit_eval(features)
110    rows = [dict(removed="(none: all features)", val_F1=full_f1, change=0.0, fit_s=full_t)]
111    for name, cols in groups.items():
112        f, s = fit_eval([c for c in features if c not in cols])
113        rows.append(dict(removed=name, val_F1=f, change=f - full_f1, fit_s=s))
114    return pd.DataFrame(rows)
